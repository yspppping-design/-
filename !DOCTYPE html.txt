<script>
// 配置区
const PASS = "anpu"; // 改这里可以换密码
const songList = [
  {name:"宝贝",src:"https://music.163.com/song/media/outer/url?id=208378.mp3"},
  {name:"关于我爱你",src:"https://music.163.com/song/media/outer/url?id=208382.mp3"},
  {name:"南国的孩子",src:"https://music.163.com/song/media/outer/url?id=208384.mp3"},
  {name:"喜欢",src:"https://music.163.com/song/media/outer/url?id=208380.mp3"},
  {name:"城市",src:"https://music.163.com/song/media/outer/url?id=208386.mp3"},
];
const lockScreen = document.getElementById('lockScreen');
const main = document.getElementById('main');
const pwdInput = document.getElementById('pwdInput');
const enterBtn = document.getElementById('enterBtn');
const audio = document.getElementById('audioPlayer');
const songWrap = document.getElementById('songList');
const noteInput = document.getElementById('noteInput');
const saveBtn = document.getElementById('saveBtn');
const noteListDom = document.getElementById('noteList');
const tags = document.querySelectorAll('.tag');
let selectedMood = "";
// 登录
enterBtn.onclick = ()=>{
  if(pwdInput.value === PASS){
    lockScreen.style.display = 'none';
    main.style.display = 'block';
    renderSongs();
    renderNotes();
  }else{
    alert("密码不对哦");
  }
}
pwdInput.onkeydown = e=>{if(e.key==='Enter') enterBtn.click();}
// 渲染歌单
function renderSongs(){
  songWrap.innerHTML = '';
  songList.forEach((item,idx)=>{
    const div = document.createElement('div');
    div.className = "song-item";
    div.innerText = item.name;
    div.onclick = ()=>{
      audio.src = item.src;
      audio.play();
      document.querySelectorAll('.song-item').forEach(s=>s.classList.remove('active'));
      div.classList.add('active');
    }
    songWrap.appendChild(div);
  })
}
// 心情标签选择
tags.forEach(t=>{
  t.onclick = ()=>{
    tags.forEach(i=>i.classList.remove('active'));
    t.classList.add('active');
    selectedMood = t.dataset.val;
  }
})
// 保存心情
saveBtn.onclick = ()=>{
  const text = noteInput.value.trim();
  if(!text) return alert("写点什么吧");
  const notes = JSON.parse(localStorage.getItem('treehole_notes') || '[]');
  notes.unshift({
    content:text,
    mood:selectedMood || "未标注",
    time:new Date().toLocaleString('zh-CN')
  })
  localStorage.setItem('treehole_notes',JSON.stringify(notes));
  noteInput.value = '';
  selectedMood = "";
  tags.forEach(t=>t.classList.remove('active'));
  renderNotes();
}
// 渲染心情时间轴
function renderNotes(){
  const notes = JSON.parse(localStorage.getItem('treehole_notes') || '[]');
  noteListDom.innerHTML = '';
  notes.forEach((n,idx)=>{
    const wrap = document.createElement('div');
    wrap.className = 'note-item';
    wrap.innerHTML = `
      <div class="note-time">${n.time}</div>
      <div class="note-tag">${n.mood}</div>
      <div class="note-content">${n.content}</div>
      <div class="del-btn" data-idx="${idx}">删除</div>
    `
    noteListDom.appendChild(wrap);
  })
  // 删除
  document.querySelectorAll('.del-btn').forEach(d=>{
    d.onclick = ()=>{
      const arr = JSON.parse(localStorage.getItem('treehole_notes') || '[]');
      arr.splice(d.dataset.idx,1);
      localStorage.setItem('treehole_notes',JSON.stringify(arr));
      renderNotes();
    }
  })
}
</script>
